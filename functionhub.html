function openIframe(title, src, icon) { document.getElementById('modal-title').innerText = title; document.getElementById('modal-icon').className = `fas ${icon}`; document.getElementById('app-iframe').src = src; document.getElementById('iframe-overlay').style.display = 'flex'; } function closeIframe() { document.getElementById('app-iframe').src = ""; document.getElementById('iframe-overlay').style.display = 'none'; } function updateTime() { const now = new Date(); document.getElementById('widget-time').innerText = now.getHours().toString().padStart(2, '0') + ":" + now.getMinutes().toString().padStart(2, '0'); const days = ['週日', '週一', '週二', '週三', '週四', '週五', '週六']; document.getElementById('widget-date').innerText = `${now.getMonth()+1}月${now.getDate()}日 ${days[now.getDay()]}`; } async function getRealWeather() { if (!navigator.geolocation) return; navigator.geolocation.getCurrentPosition(async (pos) => { try { const res = await fetch(`https://api.open-meteo.com/v1/forecast?latitude=${pos.coords.latitude}&longitude=${pos.coords.longitude}¤t_weather=true`); const data = await res.json(); document.getElementById('weather-temp').innerText = Math.round(data.current_weather.temperature) + '°C'; document.getElementById('weather-desc').innerText = "數據已同步"; const code = data.current_weather.weathercode; const iconEl = document.getElementById('weather-icon'); if (code === 0) iconEl.className = 'fas fa-sun text-yellow-400'; else if (code <= 3) iconEl.className = 'fas fa-cloud-sun text-yellow-200'; else iconEl.className = 'fas fa-cloud'; } catch (e) { document.getElementById('weather-desc').innerText = "無法更新"; } }); } window.onload = () => { renderUI(); updateTime(); setInterval(updateTime, 1000); getRealWeather(); };
